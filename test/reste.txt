void spi_test_xchgbuf(uint16_t len) {
  if (len > 2044) {  // 4 Bytes f�r Flag und L�nge!
    len = 2044;
  }
  // Test: Schreibe 2 KBytes mit einem Muster in Exchange-RAM des FPGA, lese sie zur�ck und vergleiche
  for (uint16_t i = 0; i < len; i++) {
    spi_blockbuffer.byte[i] = i & 0xFF; // Beispielmuster: 0x00, 0x01, ..., 0xFF, 0x00, ...
  }
  spi_xc_writeram(len, 0x12); // Typ 0x12
  memset(spi_blockbuffer.byte, 0, len); // Buffer leeren
  spi_xc_readram(len);
  // Vergleiche gelesene Daten mit urspr�nglichem Muster  
  bool success = true;
  for (uint16_t i = 0; i < len; i++) {
    if (spi_blockbuffer.byte[i] != (i & 0xFF)) {
      Serial.print(F("Data mismatch at index "));
      Serial.print(i);
      Serial.print(F(" = 0x"));
      Serial.print(i, HEX);
      Serial.print(F(": expected "));
      Serial.print(i & 0xFF, HEX);
      Serial.print(F(", got "));
      Serial.println(spi_blockbuffer.byte[i], HEX);
      uint16_t idx_start = (i / 16) * 16; // Zeige Dump der 16-Byte-Zeile, in der der Fehler aufgetreten ist
      spi_dump_buf(spi_blockbuffer.byte, idx_start, 16);
      success = false;
      break;
    }
  }
  if (success) {
    Serial.println(F("XCHG tested OK!"));
  } else {
    Serial.println(F("XCHG test failed!"));
  }
}


function SPI_xc_transferTest(len: Word): Boolean;
var count, idx: Word;
  hexstr: String[4];
begin
  if len > 2044 then  // 4 Bytes f?r Flag und L?nge!
    len:= 2044;
  result:= true;
  for count := 0 to len - 1 do
    BlockBuffer8[count]:= lo(count); // Fill buffer with test data
  SPI_xc_writeram(len); // Write test data to FPGA
  FillBlock(@BlockBuffer8, len, 0);
  SPI_xc_readram(len); // Read back data from FPGA
  for count := 0 to len - 1 do
    if BlockBuffer8[count] <> lo(count) then begin
      result:= false; // Data mismatch
      Serial1_sendstring('Data mismatch at address ');
      Serial1_sendIntVal(count);
      Serial1_sendstring(' = $');
      WordToHex(count, hexstr);
      Serial1_sendstring(hexstr);
      Serial1_CRLF;
      idx:= (count div 16) * 16; // Align to 16-byte boundary for dump
      SPI_dump_buf(idx, 16); // Dump 16 bytes around the mismatch
      break;
    end;
end;
